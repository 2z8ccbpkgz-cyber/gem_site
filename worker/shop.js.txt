  function strand(P0, P1, P2, r, g) {
    var len = 0, prev = qb(P0, P1, P2, 0), i, p;
    for (i = 1; i <= 40; i++) { p = qb(P0, P1, P2, i / 40); len += Math.hypot(p[0] - prev[0], p[1] - prev[1]); prev = p; }
    var n = Math.max(6, Math.round(len / (2 * r * 1.02))), s = '';
    for (i = 0; i < n; i++) { p = qb(P0, P1, P2, (i + .5) / n); s += bead(p[0], p[1], r, g); }
    return s;
  }
  function plate(it) {

    var hex = tintOf(it.n), neutral = !hex; hex = hex || (cols && cols[0]) || '#8d949b';
    var mm = parseMm(it), r = Math.max(4.6, Math.min(13, mm * 1.05)), g = 'bd' + (++UIDN), s = '';
    var nm = String(it.n).toLowerCase();
